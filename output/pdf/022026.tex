\begin{center}\LARGE\textbf{Autonomous Field Navigation of Robot}\\[2pt]\large [ID 022026]\end{center}%
\vspace{4pt}%
\noindent Develop an autonomous navigation stack for a small field robot, using LiDAR and positioning sensors for obstacle detection and safe movement.%
\vspace{6pt}%
\vspace{4pt}\hrule\vspace{8pt}%
\textbf{Objectives}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item Enable autonomous field navigation%
\item Detect obstacles using LiDAR data%
\item Fuse localization and perception signals for safer motion%
\end{itemize}%
\textbf{Tasks}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item Integrate LiDAR, GPS, and Pixhawk interfaces%
\item Build localization and navigation modules%
\item Detect obstacles from LiDAR observations%
\item Plan and execute autonomous paths in the field%
\item Validate navigation performance on the robot chassis%
\end{itemize}%
\textbf{Requirements}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item Python programming%
\item Robotics fundamentals%
\item ROS basics%
\end{itemize}%
\textbf{Equipment}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item RPLIDAR C1 LiDAR%
\item ZED{-}F9P{-}01B{-}01 GPS module%
\item Holybro Pixhawk 6X flight controller%
\item UGV Rover ROS 2 Open{-}source 6 Wheels 4WD%
\end{itemize}%
\textbf{Deliverables}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item Autonomous navigation prototype with LiDAR{-}based obstacle detection%
\item Field validation results%
\end{itemize}%
\textbf{Work Breakdown}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item 30\% hardware integration%
\item 70\% coding%
\end{itemize}%
