\begin{center}\LARGE\textbf{Vegetation and Obstacle Perception}\\[2pt]\large [ID 012026]\end{center}%
\vspace{4pt}%
\noindent Develop a perception pipeline to estimate grass height detect invasive weeds and identify field obstacles for a small ground robot.%
\vspace{6pt}%
\vspace{4pt}\hrule\vspace{8pt}%
\textbf{Objectives}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item Estimate grass height in field conditions%
\item Detect invasive weeds in camera imagery%
\item Identify poles and other obstacles around the robot%
\end{itemize}%
\textbf{Tasks}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item Build image and ranging acquisition modules%
\item Process camera and laser data to estimate grass height%
\item Detect invasive weeds and mark detections%
\item Identify poles and similar obstacles%
\item Validate performance on the robot chassis%
\end{itemize}%
\textbf{Requirements}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item Python programming%
\item Computer vision fundamentals%
\item ROS basics%
\end{itemize}%
\textbf{Equipment}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item Jetson Orin Nano%
\item Teltonika RUT240 4G LTE Router%
\item VL53L0X ranging sensors%
\item Arducam IMX219 NoIR camera%
\item UGV Rover ROS 2 Open{-}source 6 Wheels 4WD%
\end{itemize}%
\textbf{Deliverables}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item Perception prototype for grass height weeds and obstacle detection%
\item Test results from field trials%
\end{itemize}%
\textbf{Work Breakdown}%
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt,parsep=0pt,partopsep=0pt]%
\item 20\% hardware integration%
\item 80\% coding%
\end{itemize}%
